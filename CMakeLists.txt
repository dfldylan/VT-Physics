CMAKE_MINIMUM_REQUIRED(VERSION 3.11)

SET(VCPKG_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/Simulator/Thirdparty/vcpkg)
SET(CMAKE_TOOLCHAIN_FILE ${VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake CACHE FILEPATH "" FORCE)

PROJECT(VT-Physics)

ENABLE_LANGUAGE(CXX CUDA)
SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 17)
SET(CMAKE_CUDA_ARCHITECTURES 80)

IF (MSVC)
    SET(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -Xcompiler=/utf-8")
ENDIF ()

SET(VP_SOURCE_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
SET(VP_SIMULATOR_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/Simulator)
SET(VP_RUNTIME_ROOT ${VP_SIMULATOR_ROOT}/Runtime)
SET(VP_TEMPLATES_ROOT ${VP_SIMULATOR_ROOT}/Templates)
SET(VP_3RD_ROOT ${VP_SIMULATOR_ROOT}/Thirdparty)
SET(VP_LIB_ROOT ${VP_SIMULATOR_ROOT}/Library)
SET(VP_CONFIG_ROOT ${VP_SOURCE_ROOT}/ConfigTemplates)
SET(VP_BIN_ROOT ${VP_SOURCE_ROOT}/Bin)

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${VP_BIN_ROOT})

# 不在 MSVC+NVCC 上请求 OpenMP 的 CUDA 组件，默认仅启用 CXX
option(VP_ENABLE_OPENMP_CUDA "Enable OpenMP offload for CUDA (needs Clang/GCC toolchain, not MSVC+NVCC)" OFF)

if (VP_ENABLE_OPENMP_CUDA)
    find_package(OpenMP REQUIRED COMPONENTS CXX CUDA)
else()
    find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()

# 如果工程里有地方链接了 OpenMP::OpenMP_CUDA，这里做个空目标兜底，避免链接阶段报错
if (NOT TARGET OpenMP::OpenMP_CUDA)
    add_library(OpenMP::OpenMP_CUDA INTERFACE IMPORTED)
endif()

#===================================== Build Components.
ADD_SUBDIRECTORY(Simulator)
ADD_SUBDIRECTORY(Examples)

ADD_CUSTOM_TARGET(BUILD_ALL
    DEPENDS vpda vpjson vplogger vpmodel vpcore vpmanager vpugns vpdfsph vpimmct vpimm vpmct vppbf vp
)
